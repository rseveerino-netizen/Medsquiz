    function shuffle(array) {

      return [...array].sort(() => Math.random() - 0.5);

    }


    function makeQuestionBank() {

      const difficulties = ["Easy", "Intermediate", "Hard"];

      const bank = [];


      SUBJECTS.forEach((subject) => {

        for (let i = 0; i < 50; i++) {

          const template =

            QUESTION_TEMPLATES[subject][i % QUESTION_TEMPLATES[subject].length];


          bank.push({

            id: `${subject}-${i + 1}`,

            subject,

            difficulty: difficulties[i % difficulties.length],

            question: `${template.question} (Question ${i + 1})`,

            correct: template.correct,

            options: shuffle([template.correct, ...template.wrong]),

            explanation: `${template.correct} is the best answer for this ${subject.toLowerCase()} question.`

          });

        }

      });


      return bank;

    }

    function getQuestions(subject, difficulty, count) {

      const filtered = QUESTION_BANK.filter((question) => {

        const subjectMatch =

          subject === "Mixed" || question.subject === subject;


        const difficultyMatch =

          difficulty === "Mixed" || question.difficulty === difficulty;


        return subjectMatch && difficultyMatch;

      });


      return shuffle(filtered).slice(0, count);

    }

    function startSoloTimer() {

      let timeLeft = soloState.seconds;

      const fill = document.getElementById("soloTimerFill");


      fill.style.transform = "scaleX(1)";


      soloState.timerId = setInterval(() => {

        timeLeft -= 0.1;


        fill.style.transform =

          `scaleX(${Math.max(0, timeLeft / soloState.seconds)})`;


        if (timeLeft <= 0) {

          clearInterval(soloState.timerId);


          if (!soloState.answered) {

            answerSolo("", null, true);

          }

        }

      }, 100);

    }


    function answerSolo(selectedAnswer, selectedButton, timeExpired = false) {

      if (soloState.answered) return;


      soloState.answered = true;

      clearInterval(soloState.timerId);


      const question = soloState.list[soloState.index];

      const correct = selectedAnswer === question.correct;


      document.querySelectorAll("#soloAnswers .answer").forEach((button) => {

        button.disabled = true;


        if (button.textContent.includes(question.correct)) {

          button.classList.add("correct");

        }

      });


      if (!correct && selectedButton) {

        selectedButton.classList.add("incorrect");

      }


      let pointsWon = 0;


      progress.answered += 1;

      progress.bySubject[question.subject].answered += 1;


      if (correct) {

        progress.correct += 1;

        progress.bySubject[question.subject].correct += 1;

        soloState.streak += 1;


        const basePoints = {

          Easy: 80,

          Intermediate: 110,

          Hard: 150

        }[question.difficulty];


        const streakBonus = Math.min(soloState.streak * 5, 40);


        pointsWon = basePoints + streakBonus;

        progress.points += pointsWon;

      } else {

        soloState.streak = 0;


        progress.mistakes.unshift({

          subject: question.subject,

          difficulty: question.difficulty,

          question: question.question,

          correct: question.correct

        });


        progress.mistakes = progress.mistakes.slice(0, 50);

      }


      saveProgress();


      const feedback = document.getElementById("soloFeedback");

      feedback.classList.remove("hidden");


      if (correct) {

        feedback.classList.remove("wrong");

        feedback.innerHTML = `

          <strong>Correct! +${pointsWon} points</strong><br />

          ${question.explanation}

        `;

      } else {

        feedback.classList.add("wrong");

        feedback.innerHTML = `

          <strong>${timeExpired ? "Time is up." : "Incorrect answer."}</strong><br />

          Correct answer: <strong>${question.correct}</strong><br />

          ${question.explanation}

        `;

      }


      const nextButton = document.getElementById("soloNextButton");


      nextButton.textContent =

        soloState.index === soloState.list.length - 1

          ? "Finish session"

          : "Next question";


      nextButton.classList.remove("hidden");

    }


    document.getElementById("soloNextButton").addEventListener("click", () => {

      if (soloState.index < soloState.list.length - 1) {

        soloState.index += 1;

        renderSoloQuestion();

      } else {

        clearInterval(soloState.timerId);


        document.getElementById("soloQuizCard").classList.add("hidden");

        document.getElementById("soloSetup").classList.remove("hidden");


        showToast("Solo session complete. Progress saved!");

        goTo("progress");

      }

    });

      */

      const chosen = getQuestions(subject, difficulty, 20);


      if (!chosen.length) {

        showToast("No questions match your selection.");

        return;

      }


      battleState.list = [];


      for (let i = 0; i < 20; i++) {

        battleState.list.push(chosen[i % chosen.length]);

      }


      battleState.index = 0;

      battleState.currentPlayer = 0;

      battleState.scores = [0, 0];


      document.getElementById("battleP1Name").textContent =

        battleState.names[0];


      document.getElementById("battleP2Name").textContent =

        battleState.names[1];


      document.getElementById("battleSetup").classList.add("hidden");

      document.getElementById("battleQuizCard").classList.remove("hidden");


      renderBattleQuestion();

    }


    function renderBattleQuestion() {

      battleState.answered = false;


      const question = battleState.list[battleState.index];

      const turnNumber = Math.floor(battleState.index / 2) + 1;


      document.getElementById("battleP1Score").textContent =

        battleState.scores[0];


      document.getElementById("battleP2Score").textContent =

        battleState.scores[1];


      document.getElementById("battleTurnText").textContent =

        `${battleState.names[battleState.currentPlayer]}'s turn`;


      document.getElementById("battleQuestionMeta").textContent =

        `${question.subject} · ${question.difficulty}`;


      document.getElementById("battleQuestionNumber").textContent =

        `Round ${turnNumber} / ${battleState.roundsPerPlayer}`;


      document.getElementById("battleQuestionText").textContent =

        question.question;


      const answers = document.getElementById("battleAnswers");

      answers.innerHTML = "";


      const feedback = document.getElementById("battleFeedback");

      feedback.className = "feedback hidden";

      feedback.innerHTML = "";


      document.getElementById("battleNextButton").classList.add("hidden");


      question.options.forEach((option, index) => {

        const button = document.createElement("button");

        button.className = "answer";

        button.textContent = `${String.fromCharCode(65 + index)}. ${option}`;

        button.addEventListener("click", () => answerBattle(option, button));

        answers.appendChild(button);

      });

    }


    function answerBattle(selectedAnswer, selectedButton) {

      if (battleState.answered) return;


      battleState.answered = true;


      const question = battleState.list[battleState.index];

      const correct = selectedAnswer === question.correct;


      document.querySelectorAll("#battleAnswers .answer").forEach((button) => {

        button.disabled = true;


        if (button.textContent.includes(question.correct)) {

          button.classList.add("correct");

        }

      });


      if (!correct) {

        selectedButton.classList.add("incorrect");

      }


      const points = {

        Easy: 80,

        Intermediate: 110,

        Hard: 150

      }[question.difficulty];


      if (correct) {

        battleState.scores[battleState.currentPlayer] += points;

      }


      document.getElementById("battleP1Score").textContent =

        battleState.scores[0];


      document.getElementById("battleP2Score").textContent =

        battleState.scores[1];


      const feedback = document.getElementById("battleFeedback");

      feedback.classList.remove("hidden");


      if (correct) {

        feedback.classList.remove("wrong");

        feedback.innerHTML = `

          <strong>Correct! +${points} points</strong><br />

          ${question.explanation}

        `;

      } else {

        feedback.classList.add("wrong");

        feedback.innerHTML = `

          <strong>Incorrect.</strong><br />

          Correct answer: <strong>${question.correct}</strong>

        `;

      }


      document.getElementById("battleNextButton").classList.remove("hidden");

    }


    document.getElementById("battleNextButton").addEventListener("click", () => {

      battleState.index += 1;

      battleState.currentPlayer = battleState.currentPlayer === 0 ? 1 : 0;


      if (battleState.index < battleState.list.length) {

        renderBattleQuestion();

        return;

      }


      const [p1Score, p2Score] = battleState.scores;


      let result;


      if (p1Score === p2Score) {

        result = `It is a tie! ${battleState.names[0]} and ${battleState.names[1]} are co-champions.`;

      } else {

        const winner = p1Score > p2Score

          ? battleState.names[0]

          : battleState.names[1];


        result = `${winner} wins the battle!`;

      }


      document.getElementById("battleQuizCard").classList.add("hidden");

      document.getElementById("battleSetup").classList.remove("hidden");


      showToast(result);

    });


    function renderProgress() {

      const accuracy = progress.answered

        ? Math.round((progress.correct / progress.answered) * 100)

        : 0;


      document.getElementById("progressPoints").textContent = progress.points;

      document.getElementById("progressCorrect").textContent = progress.correct;

      document.getElementById("progressAccuracy").textContent =

        `${accuracy}% accuracy`;


      document.getElementById("progressMistakes").textContent =

        progress.mistakes.length;


      const rows = document.getElementById("subjectStatsRows");


      rows.innerHTML = SUBJECTS.map((subject) => {

        const subjectProgress = progress.bySubject[subject];

        const subjectAccuracy = subjectProgress.answered

          ? Math.round(

              (subjectProgress.correct / subjectProgress.answered) * 100

            )

          : 0;


        return `

          <tr>

            <td>${subject}</td>

            <td>${subjectProgress.correct}</td>

            <td>${subjectProgress.answered}</td>

            <td>${subjectAccuracy}%</td>

          </tr>

        `;

      }).join("");


      const mistakesContainer = document.getElementById("mistakesContainer");


      if (!progress.mistakes.length) {

        mistakesContainer.innerHTML =

          `<div class="empty">No mistakes saved yet. Keep going!</div>`;


        return;

      }


      mistakesContainer.innerHTML = progress.mistakes

        .slice(0, 20)

        .map((mistake) => {

          return `

            <div class="mistake-item">

              <strong>${mistake.subject} · ${mistake.difficulty}</strong>

              <div>${mistake.question}</div>

              <div class="mistake-answer">

                Correct answer: ${mistake.correct}

              </div>

            </div>

          `;

        })

        .join("");

    }


    function resetProgress() {

      const confirmReset = window.confirm(

        "Are you sure you want to erase your MedsQuiz progress?"

      );


      if (!confirmReset) return;


      progress = structuredClone(DEFAULT_PROGRESS);


      SUBJECTS.forEach((subject) => {

        progress.bySubject[subject] = { correct: 0, answered: 0 };

      });


      saveProgress();

      renderProgress();


      showToast("Saved progress has been reset.");

    }


    updateGlobalStats();
